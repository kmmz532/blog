{{- /* 本文の {{< link-card url="..." >}} をカードにする（design.md §18）。OGP が取れなければ通常のリンク */ -}} {{- $url := .Get "url" -}} {{- $ok := or (strings.HasPrefix $url "https://") (strings.HasPrefix $url "http://") -}} {{- $title := "" -}} {{- $desc := "" -}} {{- $image := "" -}} {{- if $ok -}} {{- $res := try (resources.GetRemote $url) -}} {{- if not $res.Err -}} {{- with $res.Value -}} {{- $html := .Content -}} {{- with findRESubmatch `]+property="og:title"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:title"` $html -}} {{- $title = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}} {{- end -}} {{- with findRESubmatch `]+property="og:description"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:description"` $html -}} {{- $desc = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}} {{- end -}} {{- with findRESubmatch `]+property="og:image"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:image"` $html -}} {{- $image = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}} {{- end -}} {{- end -}} {{- else -}} {{- warnf "link-card: %s: %s" $url $res.Err -}} {{- end -}} {{- end -}} {{- /* 中は span と img だけにして、

の中に入っても正しい HTML になるようにする */ -}} {{- if and $ok $title -}} {{ $title }}{{ with $desc }}{{ . }}{{ end }}{{ (urls.Parse $url).Host }}{{ with $image }}{{ end }} {{- else -}} {{ $url }} {{- end -}}
