{{- /* 記事のタイトル・概要・サムネイルを dict で返す。外部記事は front matter を優先し、空の項目だけ OGP で補う。取得に失敗したら front matter だけ */ -}} {{- $title := .Params.title | default .Title -}} {{- /* 通常記事の概要: front matter が空なら本文から作る（
 を除き、空白をまとめ、120 字で切る） */ -}}
{{- $summary := .Params.summary | default (.Content | replaceRE `(?s)` "" | plainify | replaceRE `\s+` " " | strings.TrimSpace | truncate 120 "…") -}}
{{- /* サムネイル: front matter の image（手動）が最優先。通常記事は本文の最初の画像、外部記事は OGP の画像 */ -}}
{{- $image := .Params.image | default "" -}}
{{- if .Params.external -}}
  {{- $html := "" -}}
  {{- $res := try (resources.GetRemote .Params.external) -}}
  {{- if not $res.Err -}}
    {{- with $res.Value -}}{{- $html = .Content -}}{{- end -}}
  {{- else -}}
    {{- warnf "external-meta: %s: %s" .Params.external $res.Err -}}
  {{- end -}}
  {{- $ogTitle := "" -}}
  {{- $ogDesc := "" -}}
  {{- $ogImage := "" -}}
  {{- if $html -}}
    {{- with findRESubmatch `]+property="og:title"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:title"` $html -}}
      {{- $ogTitle = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}}
    {{- end -}}
    {{- with findRESubmatch `]+property="og:description"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:description"` $html -}}
      {{- $ogDesc = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}}
    {{- end -}}
    {{- with findRESubmatch `]+property="og:image"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:image"` $html -}}
      {{- $ogImage = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}}
    {{- end -}}
  {{- end -}}
  {{- $title = .Params.title | default $ogTitle | default .Params.external -}}
  {{- $summary = .Params.summary | default $ogDesc -}}
  {{- $image = $image | default $ogImage -}}
{{- else -}}
  {{- if not $image -}}
    {{- with findRESubmatch `]+src="([^"]+)"` .Content -}}
      {{- $image = index (index . 0) 1 -}}
    {{- end -}}
  {{- end -}}
{{- end -}}
{{- return (dict "title" $title "summary" $summary "image" $image) -}}
