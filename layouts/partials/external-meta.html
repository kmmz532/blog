{{- /* 記事のタイトル・概要・サムネイルを dict で返す。外部記事は front matter を優先し、空の項目だけ OGP で補う。取得に失敗したら front matter だけ */ -}} {{- $title := .Params.title | default .Title -}} {{- /* 通常記事の概要: front matter が空なら本文から作る（見出し・コード・リンクカードを除外） */ -}} {{- $summaryBody := .Content | replaceRE `(?s)]*class="[^"]*link-card[^"]*"[^>]*>.*?` "" | replaceRE `(?s)]*>.*?` "" | replaceRE `(?s)]*>.*?` "" | replaceRE `\s*]*>` "、" | replaceRE `]*>|` "" | replaceRE `(?i)` " " -}} {{- $summary := .Params.summary | default ($summaryBody | plainify | replaceRE `\s+` " " | strings.TrimSpace | truncate 120 "…") -}} {{- /* サムネイル: front matter の image（手動）が最優先。通常記事は本文の最初の画像、外部記事は OGP の画像 */ -}} {{- $image := .Params.image | default "" -}} {{- /* 編集画面のサムネイル候補（https の画像。外部記事は OGP の og:image と twitter:image を全部集める。通常記事は空） */ -}} {{- $images := slice -}} {{- if .Params.external -}} {{- $html := "" -}} {{- $res := try (resources.GetRemote .Params.external) -}} {{- if not $res.Err -}} {{- with $res.Value -}}{{- $html = .Content -}}{{- end -}} {{- else -}} {{- warnf "external-meta: %s: %s" .Params.external $res.Err -}} {{- end -}} {{- $ogTitle := "" -}} {{- $ogDesc := "" -}} {{- $ogImage := "" -}} {{- if $html -}} {{- with findRESubmatch `]+property="og:title"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:title"` $html -}} {{- $ogTitle = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}} {{- end -}} {{- with findRESubmatch `]+property="og:description"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:description"` $html -}} {{- $ogDesc = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}} {{- end -}} {{- with findRESubmatch `]+property="og:image"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+property="og:image"` $html -}} {{- $ogImage = htmlUnescape (or (index (index . 0) 1) (index (index . 0) 2)) -}} {{- end -}} {{- with findRESubmatch `]+(?:property|name)="(?:og|twitter):image"[^>]+content="([^"]*)"|]+content="([^"]*)"[^>]+(?:property|name)="(?:og|twitter):image"` $html -}} {{- range . -}} {{- $u := htmlUnescape (or (index . 1) (index . 2)) -}} {{- if hasPrefix $u "https://" -}}{{- $images = $images | append $u -}}{{- end -}} {{- end -}} {{- end -}} {{- $images = $images | uniq -}} {{- end -}} {{- $title = .Params.title | default $ogTitle | default .Params.external -}} {{- $summary = .Params.summary | default $ogDesc -}} {{- $image = $image | default $ogImage -}} {{- else -}} {{- if not $image -}} {{- with findRESubmatch `]+src="([^"]+)"` .Content -}} {{- $image = index (index . 0) 1 -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "title" $title "summary" $summary "image" $image "images" $images) -}}
